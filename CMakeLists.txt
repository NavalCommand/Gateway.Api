cmake_minimum_required(VERSION 4.2)
project(Gateway.Api VERSION 0.1.0 LANGUAGES CXX)
include(cmake/Common.cmake)
set(GATEWAY_API_PROTO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/proto" CACHE INTERNAL "")
set(gen "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${gen}")
set(names session orders world incidents)
foreach(name IN LISTS names)
 set(path "navalcommand/gateway/v1/${name}")
 add_custom_command(OUTPUT "${gen}/${path}.pb.cc" "${gen}/${path}.pb.h"
 COMMAND protobuf::protoc --proto_path=${GATEWAY_API_PROTO_DIR} --cpp_out=${gen} ${GATEWAY_API_PROTO_DIR}/${path}.proto
 DEPENDS "${GATEWAY_API_PROTO_DIR}/${path}.proto" protobuf::protoc)
 list(APPEND sources "${gen}/${path}.pb.cc")
endforeach()
add_library(Gateway.Api STATIC ${sources} src/NavalCommand/Gateway/Api/Codec.cpp)
target_include_directories(Gateway.Api PUBLIC public "${gen}")
target_link_libraries(Gateway.Api PUBLIC protobuf::libprotobuf)
add_executable(schema-roundtrip app/SchemaRoundtrip.cpp)
target_link_libraries(schema-roundtrip PRIVATE Gateway.Api)
if(PROJECT_IS_TOP_LEVEL)
 enable_testing()
 add_executable(api-tests test/CodecTest.cpp)
 target_link_libraries(api-tests PRIVATE Gateway.Api GTest::gtest_main)
 add_test(NAME codec COMMAND api-tests)
endif()
